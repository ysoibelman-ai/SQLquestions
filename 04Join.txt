SELECT customerName, AVG(amount)
FROM customers
JOIN payments
ON customers.customerNumber = payments.customerNumber
GROUP BY customerName

3
SELECT c.customerName, AVG(amount)
FROM customers as c
JOIN payments as p
ON c.customerNumber = p.customerNumber
GROUP BY (customerName)

4

ON c.customerNumber = p.customerNumber
